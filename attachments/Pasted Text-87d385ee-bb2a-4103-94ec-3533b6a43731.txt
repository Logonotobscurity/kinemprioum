JS_CAROUSEL = """
(function initCarousel(){
  const track = document.getElementById('dungTrack');
  const dots  = document.getElementById('dungDots');
  const prev  = document.getElementById('dungPrev');
  const next  = document.getElementById('dungNext');
  if(!track) return;
  track.innerHTML = DUNGEONS.map(buildDungeonCard).join('');
  const CARD_W = 296;
  const visible  = () => Math.max(1, Math.floor(track.parentElement.offsetWidth / CARD_W));
  const maxIdx   = () => Math.max(0, DUNGEONS.length - visible());
  let idx = 0;
  function buildDots(){
    const pages = maxIdx() + 1;
    dots.innerHTML = '';
    for(let i=0;i<pages;i++){
      const d = document.createElement('div');
      d.className = 'dungeon-dot' + (i===idx?' active':'');
      d.onclick = () => goTo(i);
      dots.appendChild(d);
    }
  }
  function goTo(i){
    idx = Math.max(0, Math.min(i, maxIdx()));
    track.style.transform = `translateX(-${idx * CARD_W}px)`;
    dots.querySelectorAll('.dungeon-dot').forEach((d,j) =>
      d.classList.toggle('active', j===idx));
  }
  prev.onclick = () => goTo(idx - 1);
  next.onclick = () => goTo(idx + 1);
  let auto = setInterval(() => goTo(idx >= maxIdx() ? 0 : idx + 1), 4800);
  track.parentElement.addEventListener('mouseenter', () => clearInterval(auto));
  track.parentElement.addEventListener('mouseleave', () => {
    auto = setInterval(() => goTo(idx >= maxIdx() ? 0 : idx + 1), 4800);
  });
  let tx = 0;
  track.addEventListener('touchstart', e => { tx = e.touches[0].clientX; }, {passive:true});
  track.addEventListener('touchend', e => {
    const dx = tx - e.changedTouches[0].clientX;
    if(Math.abs(dx) > 40) goTo(dx > 0 ? idx+1 : idx-1);
  }, {passive:true});
  buildDots();
  window.addEventListener('resize', () => { buildDots(); goTo(Math.min(idx, maxIdx())); });
})();
"""
